! =====================================================
! R1 Configuration - Packet Tracer Static Routing Lab
! =====================================================

enable
configure terminal

hostname R1

! --- LAN interface (toward PC1 / Switch1) ---
interface GigabitEthernet0/1
 ip address 192.168.1.254 255.255.255.0
 no shutdown
 exit

! --- WAN interface (toward R2) ---
interface GigabitEthernet0/0
 ip address 192.168.12.1 255.255.255.0
 no shutdown
 exit

! --- Static routes to remote networks ---
ip route 192.168.13.0 255.255.255.0 192.168.12.2
ip route 192.168.3.0 255.255.255.0 192.168.12.2

! --- Cosmetic / good practice ---
no ip domain-lookup
line con 0
 logging synchronous
 exit

end
write memory
